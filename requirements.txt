def generate_recommendations(
    result,
    gpu=False,
    rwx=False,
    network="25Gb",
    storage="OpenShift Data Foundation"
):

    recommendations = []

    ##################################################
    # Arquitectura
    ##################################################

    if result["workers"] < 5:
        recommendations.append(
            "Arquitectura Compact recomendada para ambientes pequeños."
        )

    elif result["workers"] < 15:
        recommendations.append(
            "Arquitectura Standard recomendada."
        )

    else:
        recommendations.append(
            "Arquitectura Enterprise recomendada."
        )

    ##################################################
    # Storage
    ##################################################

    if storage == "OpenShift Data Foundation":
        recommendations.append(
            "Utilizar OpenShift Data Foundation para almacenamiento persistente."
        )

    else:
        recommendations.append(
            f"Validar compatibilidad CSI con {storage}."
        )

    ##################################################
    # RWX
    ##################################################

    if rwx:
        recommendations.append(
            "Habilitar almacenamiento RWX para Live Migration."
        )

    ##################################################
    # GPU
    ##################################################

    if gpu:
        recommendations.append(
            "Configurar GPU Passthrough o NVIDIA GPU Operator."
        )

    ##################################################
    # Networking
    ##################################################

    if network == "10Gb":
        recommendations.append(
            "Se recomienda 25Gb para cargas de producción."
        )

    elif network == "25Gb":
        recommendations.append(
            "25Gb es una excelente opción para OpenShift Virtualization."
        )

    elif network == "40Gb":
        recommendations.append(
            "40Gb recomendado para ambientes de alta demanda."
        )

    elif network == "100Gb":
        recommendations.append(
            "100Gb recomendado para IA, HPC o grandes volúmenes de datos."
        )

    ##################################################
    # Alta disponibilidad
    ##################################################

    recommendations.append(
        "Implementar tres nodos Control Plane para Alta Disponibilidad."
    )

    recommendations.append(
        "Configurar monitoreo con Prometheus y Grafana."
    )

    recommendations.append(
        "Implementar Backup con OpenShift API for Data Protection."
    )

    recommendations.append(
        "Utilizar GitOps (ArgoCD) para automatización."
    )

    recommendations.append(
        "Implementar Red Hat Advanced Cluster Management."
    )

    recommendations.append(
        "Implementar Red Hat Advanced Cluster Security."
    )

    recommendations.append(
        "Utilizar Quay Enterprise como registro de imágenes."
    )

    return recommendations
